void updateRelays() {
  // Инвертируйте LOW/HIGH здесь, если ваши реле управляются низким уровнем (LOW)
  digitalWrite(PIN_ELECTROLYZER, electroState ? HIGH : LOW);
  digitalWrite(PIN_IGNITION, ignitionState ? HIGH : LOW);
  digitalWrite(PIN_VALVE, valveState ? HIGH : LOW);
}

void setup() {
  Serial.begin(115200);
  
  pinMode(PIN_ELECTROLYZER, OUTPUT);
  pinMode(PIN_IGNITION, OUTPUT);
  pinMode(PIN_VALVE, OUTPUT);
  updateRelays(); // Применяем начальные состояния (всё выключено)

  WiFi.begin(ssid, password);
  Serial.print("Подключение к WiFi");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi подключен! IP адрес: ");
  Serial.println(WiFi.localIP());

  // Маршрут для главной страницы
  server.on("/", []() {
    server.send(200, "text/html", index_html);
  });

  // Маршрут для отдачи статуса (JSON)
  server.on("/status", []() {
    // Чтение датчика давления (0-4095 для ESP32). 
    // Замените математику ниже на вашу формулу калибровки давления!
    int rawPressure = analogRead(PIN_PRESSURE);
    float pressureBar = (rawPressure / 4095.0) * 10.0; // Пример: 0-10 Bar

    String json = "{";
    json += "\"systemOn\":" + String(systemOn ? "true" : "false") + ",";
    json += "\"electro\":" + String(electroState ? "true" : "false") + ",";
    json += "\"ignition\":" + String(ignitionState ? "true" : "false") + ",";
    json += "\"valve\":" + String(valveState ? "true" : "false") + ",";
    json += "\"pressure\":" + String(pressureBar, 1) + ",";
    json += "\"shuttingDown\":" + String(isShuttingDown ? "true" : "false");
    json += "}";
    server.send(200, "application/json", json);
  });

  // Обычное переключение устройств
  server.on("/toggle", []() {
    if (server.hasArg("dev")) {
      String dev = server.arg("dev");
      if (dev == "power") {
        systemOn = !systemOn;
        if (!systemOn) {
          // Если выключаем обычным способом, просто рубим всё
          electroState = false;
          ignitionState = false;
          valveState = false;
        }
      } 
      else if (systemOn && !isShuttingDown) { 
        // Разрешаем управлять остальным, только если система включена
        if (dev == "electro") electroState = !electroState;
        if (dev == "ignition") ignitionState = !ignitionState;
        if (dev == "valve") valveState = !valveState;
      }
      updateRelays();
    }
    server.send(200, "text/plain", "OK");
  });

  // Последовательность при двойном клике (Безопасное выключение)
  server.on("/shutdown", []() {
    if(systemOn && !isShuttingDown) {
      electroState = false;   // Выключаем электролизер
      ignitionState = false;  // Выключаем поджиг
      valveState = true;      // ОТКРЫВАЕМ клапан
      updateRelays();

      isShuttingDown = true;
      shutdownStartTime = millis(); // Засекаем время
    }
    server.send(200, "text/plain", "SHUTDOWN_STARTED");
  });

  server.begin();
}

void loop() {
  server.handleClient();

  // Обработка неблокирующего таймера выключения
  if (isShuttingDown) {
    if (millis() - shutdownStartTime >= SHUTDOWN_DURATION) {
      // 5 секунд прошло
      valveState = false; // Закрываем клапан
      systemOn = false;   // Полностью выключаем систему
      isShuttingDown = false; // Завершаем режим выключения
      
      updateRelays();
      Serial.println("Система безопасно выключена.");
    }
  }
}

### Как это работает:

-   **Секрет двойного клика:** В JavaScript используется функция `setTimeout`. При первом клике скрипт ждет 300 миллисекунд. Если за это время происходит второй клик, таймер сбрасывается и вызывается функция `initiateShutdown()`. Если второго клика нет — срабатывает обычное Вкл/Выкл.
    
-   **Датчик давления:** Я прописал чтение порта `PIN_PRESSURE (34)` для вывода. Пока там стоит примерная формула (перевод значения 0-4095 в 0-10 Bar). Вам нужно будет подставить туда коэффициенты вашего реального датчика.
    
-   **Безопасное выключение (`millis()`):** Когда вызывается маршрут `/shutdown`, микроконтроллер мгновенно рубит электролизер и поджиг, открывает клапан и ставит флаг `isShuttingDown = true`, запоминая текущее время. Цикл `loop()` продолжает свободно крутиться, обеспечивая работу веб-сервера, и ровно через 5000 мс закрывает клапан и гасит статус системы.

  digitalWrite(pinValve, valveNeeded ? HIGH : LOW);
  handlePiezoTimer();
}

void processLogic() {
unsigned long currentMillis = millis();
// 1. Блокировка - Высший приоритет
if (isLocked || !systemOn) {
digitalWrite(pinElectrolyzer, LOW);
digitalWrite(pinPiezo, LOW);
digitalWrite(pinValve, LOW);
resetProcesses();
return;
}
bool electrolyzerNeeded = false;
bool valveNeeded = false;
// 2. Режим: Постоянная подача газа
if (constGasActive) {
if (currentPressure < 1.0) {
electrolyzerNeeded = true;
valveNeeded = true;
if ((currentMillis - constGasStartTime >= timeConstGasPiezoDelay) && !constGasPiezoFired) {
startPiezoTimer();
constGasPiezoFired = true;
}
} else {
constGasStartTime = currentMillis;
constGasPiezoFired = false;
}
}
// 3. Режим: Репульсор
if (repulsorActive) {
unsigned long repulsorElapsed = currentMillis - repulsorStartTime;
if (repulsorElapsed < timeRepulsorValve) {
valveNeeded = true;
if (repulsorElapsed >= timeRepulsorPiezoDelay && !piezoActive) {
startPiezoTimer();
}
} else {
repulsorActive = false;
}
}
// 4. Режим: Ручная подача газа
if (manualGasSupplyActive) {
valveNeeded = true;
electrolyzerNeeded = true;
}
// 5. Обычное поддержание давления
if (!constGasActive && !manualGasSupplyActive) {
if (currentPressure < targetPressure) {
electrolyzerNeeded = true;
}
}
// --- Применение состояний к реле ---
digitalWrite(pinElectrolyzer, electrolyzerNeeded ? HIGH : LOW);
digitalWrite(pinValve, valveNeeded ? HIGH : LOW);
// --- Независимая обработка таймера пьезоподжига ---
handlePiezoTimer();
}
// === ФУНКЦИИ ПЬЕЗОПОДЖИГА (NON-BLOCKING) ===
void startPiezoTimer() {
if (!piezoActive) {
piezoActive = true;
piezoStartTime = millis();
digitalWrite(pinPiezo, HIGH);
}
}
void handlePiezoTimer() {
if (piezoActive) {
if (millis() - piezoStartTime >= timePiezoWork) {
piezoActive = false;
digitalWrite(pinPiezo, LOW);
}
}
}

      executeCommand(id);
    }
    request->send(200, "text/plain", "OK");
  });

    json += "\"pwr\":" + String(systemOn ? 1 : 0) + ",";
    json += "\"lck\":" + String(isLocked ? 1 : 0) + ",";
    json += "\"mod\":" + String(targetPressure) + ",";
    json += "\"cg\":" + String(constGasActive ? 1 : 0) + ",";
    json += "\"mg\":" + String(manualGasSupplyActive ? 1 : 0) + ",";
    json += "\"pz\":" + String(piezoActive ? 1 : 0) + ",";

    json += "\"purge\":" + String(isSafeShuttingDown ? 1 : 0);
    json += "}";
    request->send(200, "application/json", json);
  });

  server.begin();

void loop() {
  readPressure();
  checkLocalButtonsDebounced();
  checkRemoteData();
  updateIndication();
  processLogic();
}

void checkLocalButtonsDebounced() {
  unsigned long currentMillis = millis();
  for (int i = 0; i < NUM_BUTTONS; i++) {
    int reading = digitalRead(btnPins[i]);
    if (reading != lastBtnStates[i]) lastDebounceTime[i] = currentMillis;
    if ((currentMillis - lastDebounceTime[i]) > debounceDelay) {
      if (reading != btnStates[i]) {
        btnStates[i] = reading;
        if (btnStates[i] == LOW) executeCommand(btnCommands[i]);
      }
    }
    lastBtnStates[i] = reading;
  }
}

void executeCommand(byte cmd) {
  // Если идет продувка, команды не принимаются (кроме экстренных)
  if (isSafeShuttingDown) return;

  switch (cmd) {
    case 1: systemOn = !systemOn; break;
    case 2: isLocked = !isLocked; break;
    case 3: 
      constGasActive = !constGasActive; 
      if (constGasActive) { constGasStartTime = millis(); constGasPiezoFired = false; }
      break;
    case 4: startPiezoTimer(); break;
    case 5: if (currentPressure >= 3.9) { repulsorActive = true; repulsorStartTime = millis(); } break;
    case 6: targetPressure = (targetPressure == 1) ? 4 : 1; break;
    case 7: manualGasSupplyActive = !manualGasSupplyActive; break;
    case 99: // СИГНАЛ БЕЗОПАСНОГО ВЫКЛЮЧЕНИЯ
      if (systemOn) {
        isSafeShuttingDown = true;
        purgeStartTime = millis();
        resetProcesses(); 
      }
      break;
  }
}

void processLogic() {
  unsigned long currentMillis = millis();

  // --- ЛОГИКА БЕЗОПАСНОГО ВЫКЛЮЧЕНИЯ (ПРОДУВКА) ---
  if (isSafeShuttingDown) {
    digitalWrite(pinElectrolyzer, LOW);
    digitalWrite(pinPiezo, LOW);
    if (currentMillis - purgeStartTime < purgeDuration) {
      digitalWrite(pinValve, HIGH); // Открываем клапан на 5 секунд
    } else {
      digitalWrite(pinValve, LOW);
      isSafeShuttingDown = false;
      systemOn = false; // После продувки окончательно выключаем
    }
    return; // Остальную логику пропускаем
  }

  // --- ОБЫЧНАЯ ЛОГИКА ---
  if (isLocked || !systemOn) {
    digitalWrite(pinElectrolyzer, LOW);
    digitalWrite(pinPiezo, LOW);
    digitalWrite(pinValve, LOW);
    resetProcesses(); 
    return;
  }

  bool electrolyzerNeeded = false;
  bool valveNeeded = false;

  if (constGasActive) {
    if (currentPressure < 1.0) {
      electrolyzerNeeded = true;
      valveNeeded = true;
      if ((currentMillis - constGasStartTime >= timeConstGasPiezoDelay) && !constGasPiezoFired) {
        startPiezoTimer();
        constGasPiezoFired = true; 
      }
    } else {
      constGasStartTime = currentMillis; constGasPiezoFired = false;
    }
  }

  if (repulsorActive) {
    unsigned long repElapsed = currentMillis - repulsorStartTime;
    if (repElapsed < timeRepulsorValve) {
      valveNeeded = true;
      if (repElapsed >= timeRepulsorPiezoDelay && !piezoActive) startPiezoTimer();
    } else { repulsorActive = false; }
  }

  if (manualGasSupplyActive) { valveNeeded = true; electrolyzerNeeded = true; }

  if (!constGasActive && !manualGasSupplyActive) {
    if (currentPressure < targetPressure) electrolyzerNeeded = true;
  }

  digitalWrite(pinElectrolyzer, electrolyzerNeeded ? HIGH : LOW);
  digitalWrite(pinValve, valveNeeded ? HIGH : LOW);
  handlePiezoTimer();
}

void startPiezoTimer() {
  if (!piezoActive && !isSafeShuttingDown) {
    piezoActive = true;
    piezoStartTime = millis();
    digitalWrite(pinPiezo, HIGH);
  }
}

void handlePiezoTimer() {
  if (piezoActive) {
    if (millis() - piezoStartTime >= timePiezoWork) {
      piezoActive = false;
      digitalWrite(pinPiezo, LOW);
    }
  }
}

void readPressure() {
  int raw = analogRead(pinPressure);
  // Калибровка: 0..4095 -> 0..12 Bar (подправь под свой датчик если нужно)
  currentPressure = (raw / 4095.0) * 12.0; 
}

void checkRemoteData() {
  if (radio.available()) {
    radio.read(&dataIn, sizeof(PacketData));
    executeCommand(dataIn.btnId);
  }
}

void updateIndication() {
  digitalWrite(ledPower, systemOn ? HIGH : LOW);
  digitalWrite(ledLock, isLocked ? HIGH : LOW);
  digitalWrite(ledMode, (targetPressure == 4) ? HIGH : LOW);
}

void resetProcesses() {
  piezoActive = false;
  repulsorActive = false;
  constGasActive = false;
  manualGasSupplyActive = false;
  constGasPiezoFired = false;
}

void setup() {
  Serial.begin(115200);
  pinMode(pinElectrolyzer, OUTPUT); digitalWrite(pinElectrolyzer, LOW);
  pinMode(pinValve, OUTPUT);        digitalWrite(pinValve, LOW);
  pinMode(pinPiezo, OUTPUT);        digitalWrite(pinPiezo, LOW);
  pinMode(ledPower, OUTPUT); pinMode(ledLock, OUTPUT); pinMode(ledMode, OUTPUT);

  for (int i = 0; i < NUM_BUTTONS; i++) {
    pinMode(btnPins[i], INPUT_PULLUP);
    btnStates[i] = HIGH; lastBtnStates[i] = HIGH;
  }

  if (radio.begin()) {
    radio.openReadingPipe(0, address);
    radio.setPALevel(RF24_PA_MAX);
    radio.startListening();
  }

  WiFi.softAP(ap_ssid, ap_password);
  server.on("/", HTTP_GET, [](AsyncWebServerRequest *request){ request->send_P(200, "text/html", index_html); });
  server.on("/cmd", HTTP_GET, [](AsyncWebServerRequest *request){
    if (request->hasParam("id")) executeCommand(request->getParam("id")->value().toInt());
    request->send(200, "text/plain", "OK");
  });
  server.on("/status", HTTP_GET, [](AsyncWebServerRequest *request){
    String j = "{";
    j += "\"p\":" + String(currentPressure) + ",\"pwr\":" + (systemOn?1:0) + ",\"lck\":" + (isLocked?1:0);
    j += ",\"mod\":" + String(targetPressure) + ",\"cg\":" + (constGasActive?1:0) + ",\"mg\":" + (manualGasSupplyActive?1:0);
    j += ",\"pz\":" + (piezoActive?1:0) + ",\"rp\":" + (repulsorActive?1:0) + ",\"purge\":" + (isPurging?1:0) + "}";
    request->send(200, "application/json", j);
  });
  server.begin();
}

void loop() {
  readPressure();
  checkLocalButtonsDebounced();
  checkRemoteData();
  updateIndication();
  processLogic();
}

void checkLocalButtonsDebounced() {
  unsigned long m = millis();
  for (int i = 0; i < NUM_BUTTONS; i++) {
    int r = digitalRead(btnPins[i]);
    if (r != lastBtnStates[i]) lastDebounceTime[i] = m;
    if ((m - lastDebounceTime[i]) > debounceDelay) {
      if (r != btnStates[i]) {
        btnStates[i] = r;
        if (btnStates[i] == LOW) executeCommand(btnCommands[i]);
      }
    }
    lastBtnStates[i] = r;
  }
}

void executeCommand(byte cmd) {
  if (isPurging) return; // Игнорируем команды во время продувки

  switch (cmd) {
    case 1: systemOn = !systemOn; break;
    case 2: isLocked = !isLocked; break;
    case 3: constGasActive = !constGasActive; if (constGasActive) { constGasStartTime = millis(); constGasPiezoFired = false; } break;
    case 4: startPiezoTimer(); break;
    case 5: if (currentPressure >= 3.9) { repulsorActive = true; repulsorStartTime = millis(); } break;
    case 6: targetPressure = (targetPressure == 1) ? 4 : 1; break;
    case 7: manualGasSupplyActive = !manualGasSupplyActive; break;
    case 99: // БЕЗОПАСНОЕ ВЫКЛЮЧЕНИЕ
      if (systemOn) {
        isPurging = true;
        purgeStartTime = millis();
        resetProcesses(); // Сбрасываем все флаги
      }
      break;
  }
}

void processLogic() {
  unsigned long currentMillis = millis();

  // 1. ЛОГИКА ПРОДУВКИ ПРИ ВЫКЛЮЧЕНИИ (Высший приоритет)
  if (isPurging) {
    digitalWrite(pinElectrolyzer, LOW);
    digitalWrite(pinPiezo, LOW);
    if (currentMillis - purgeStartTime < timePurgeBeforeOff) {
      digitalWrite(pinValve, HIGH);
    } else {
      digitalWrite(pinValve, LOW);
      isPurging = false;
      systemOn = false; // Окончательно выключаем систему
    }
    return;
  }

  // 2. Блокировка
  if (isLocked || !systemOn) {
    digitalWrite(pinElectrolyzer, LOW);
    digitalWrite(pinPiezo, LOW);
    digitalWrite(pinValve, LOW);
    resetProcesses();
    return;
  }

  bool elNeeded = false;
  bool valveNeeded = false;

  if (constGasActive) {
    if (currentPressure < 1.0) {
      elNeeded = true; valveNeeded = true;
      if ((currentMillis - constGasStartTime >= timeConstGasPiezoDelay) && !constGasPiezoFired) {
        startPiezoTimer(); constGasPiezoFired = true;
      }
    } else { constGasStartTime = currentMillis; constGasPiezoFired = false; }
  }

  if (repulsorActive) {
    unsigned long elap = currentMillis - repulsorStartTime;
    if (elap < timeRepulsorValve) {
      valveNeeded = true;
      if (elap >= timeRepulsorPiezoDelay && !piezoActive) startPiezoTimer();
    } else { repulsorActive = false; }
  }

  if (manualGasSupplyActive) { valveNeeded = true; elNeeded = true; }

  if (!constGasActive && !manualGasSupplyActive) {
    if (currentPressure < targetPressure) elNeeded = true;
  }

  digitalWrite(pinElectrolyzer, elNeeded ? HIGH : LOW);
  digitalWrite(pinValve, valveNeeded ? HIGH : LOW);
  handlePiezoTimer();
}

void startPiezoTimer() {
  if (!piezoActive && !isPurging) {
    piezoActive = true;
    piezoStartTime = millis();
    digitalWrite(pinPiezo, HIGH);
  }
}

void handlePiezoTimer() {
  if (piezoActive) {
    if (millis() - piezoStartTime >= timePiezoWork) {
      piezoActive = false;
      digitalWrite(pinPiezo, LOW);
    }
  }
}

void readPressure() {
  int raw = analogRead(pinPressure);
  currentPressure = (raw / 4095.0) * 12.0; 
}

void checkRemoteData() {
  if (radio.available()) {
    radio.read(&dataIn, sizeof(PacketData));
    executeCommand(dataIn.btnId);
  }
}

void updateIndication() {
  digitalWrite(ledPower, (systemOn || isPurging) ? HIGH : LOW);
  digitalWrite(ledLock, isLocked ? HIGH : LOW);
  digitalWrite(ledMode, (targetPressure == 4) ? HIGH : LOW);
}

void resetProcesses() {
  piezoActive = false; repulsorActive = false;
  constGasActive = false; manualGasSupplyActive = false;
  constGasPiezoFired = false;
}

### Главные изменения:

1.  **Функция `isPurging`**: Если вы дважды быстро нажали "POWER" в браузере, ESP32 переходит в состояние продувки. В этом режиме переменная `systemOn` еще `true`, но вся работа электролиза и поджига прекращена, а клапан принудительно открыт.
    
2.  **JSON статус**: Добавлено поле `"purge"`. Теперь сайт видит, когда идет сброс давления, и подсвечивает индикатор **PURGING** голубым цветом с анимацией пульсации.
    
3.  **Безопасность**: Пока идет 5-секундная продувка, контроллер игнорирует любые другие команды, чтобы гарантированно завершить цикл очистки.
    
4.  **Давление**: Значение выводится в центр экрана, чтобы вы могли видеть, как оно падает во время продувки.

Вот основные причины, почему строчек становится меньше:

### 1\. Объединение логики (DRY — Don't Repeat Yourself)

В первых версиях кода часто встречаются повторяющиеся блоки. Например, вместо того чтобы в пяти разных местах писать `digitalWrite(pin, LOW)`, я создаю одну функцию `resetProcesses()` или группирую проверки в один блок `if/else`. Это уменьшает объем кода и снижает риск ошибок.

### 2\. Использование тернарных операторов

Вместо длинной конструкции:

C++

```
if (systemOn) {
  json += "1";
} else {
  json += "0";
}
```

Я использую краткую запись: `json += systemOn ? "1" : "0";`. Это экономит 4 строчки за раз, не теряя в функциональности.

void executeCommand(byte cmd) {
  if (isPurging && cmd != 2) return;
  switch (cmd) {
    case 1: systemOn = !systemOn; break;
    case 2: isLocked = !isLocked; break;
    case 3: constGasActive = !constGasActive; constGasStartTime = millis(); break;
    case 4: startPiezo(); break;
    case 6: targetPressure = (targetPressure == 1) ? 4 : 1; break;
    case 7: manualGasSupplyActive = !manualGasSupplyActive; break;
    case 99: if(systemOn) { isPurging = true; purgeStartTime = millis(); resetProcesses(); } break;
  }
}

  currentPressure = (analogRead(A0) / 1023.0) * 12.0; 

  // Кнопка Power (D0) с двойным кликом
  static bool lastBtn = HIGH;
  bool btn = digitalRead(pinButton);
  if (btn == LOW && lastBtn == HIGH) {

    if (now - lastClickTime < doubleClickGap) executeCommand(99);
    else executeCommand(1);
    lastClickTime = now;
    delay(50); // Простейший антидребезг

  // Радио
  if (radio.available()) { radio.read(&dataIn, sizeof(dataIn)); executeCommand(dataIn.btnId); }

  // Логика
  unsigned long now = millis();
  if (isPurging) {
    digitalWrite(pinElectrolyzer, LOW);
    if (now - purgeStartTime < 5000) digitalWrite(pinValve, HIGH);
    else { digitalWrite(pinValve, LOW); isPurging = false; systemOn = false; }
  } else if (isLocked || !systemOn) {
    digitalWrite(pinElectrolyzer, LOW); digitalWrite(pinValve, LOW); digitalWrite(pinPiezo, LOW);
    resetProcesses();
  } else {
    bool e = false, v = false;
    if (constGasActive) {
      if (currentPressure < 1.0) { e = v = true; }
    } else if (manualGasSupplyActive) { e = v = true; }
    else if (currentPressure < targetPressure) { e = true; }
    
    digitalWrite(pinElectrolyzer, e);
    digitalWrite(pinValve, v);
  }

  if (piezoActive && now - piezoStartTime > 500) { piezoActive = false; digitalWrite(pinPiezo, LOW); }
}

Давай разберем, что будет работать, а что — под вопросом:

### 1\. Что точно заработает:

-   **Прошивка:** Через USB-кабель код зальется без проблем.
    
-   **Wi-Fi точка доступа:** Плата создаст сеть, ты сможешь зайти на страницу управления с телефона.
    
-   **Логика кода:** Ты сможешь увидеть в «Мониторе порта» (Serial Monitor), как меняются переменные и приходят команды.
    

### 2\. С чем могут быть проблемы (Питание):

USB-порт компьютера выдает максимум **500 мА**.

-   **ESP8266 + Wi-Fi:** В моменты передачи данных по Wi-Fi чип потребляет до 200–300 мА.
    
-   **NRF24L01:** Если он начнет активно искать пульт, это еще +100 мА.
    
-   **Реле:** Это самый опасный момент. Каждое включенное реле потребляет около 70–100 мА. Если у тебя одновременно включатся 3 реле + Wi-Fi, ток превысит лимит USB.
    

**Результат:** Плата просто «мигнет» и перезагрузится (Brown-out reset).

### 3\. Датчик давления (ВНИМАНИЕ!)

Это самый важный момент для твоей NodeMCU (на фото):

-   Пин **A0** на этой плате понимает максимум **3.3В** (на некоторых версиях NodeMCU стоит делитель до 1В, но чаще 3.3В).
    
-   Большинство промышленных датчиков давления работают от **5В** и выдают на выходе до **4.5В**.
    
-   **Если ты подашь 5В напрямую на A0, аналоговый вход сгорит.**
    

**Решение для теста:** Если датчика пока нет, просто оставь пин A0 никуда не подключенным (будут случайные помехи) или замкни его на GND (будет 0 bar).

### 4\. Подключение реле

Если ты подключаешь модуль реле (синие блоки, как на краю твоего фото):

-   Не питай их от вывода `3V3` на плате ESP8266.
    
-   Подключи их питание (VCC) к пину `Vin` (там будет 5В от USB). Но помни про лимит тока USB-порта!
    

### Итог:

Для **первичной проверки прошивки и веб-интерфейса** — ДА, подключай через комп. Для **полноценного теста** с щелканьем реле и датчиком — лучше использовать внешний блок питания (например, зарядку от смартфона на 2А, подключенную к USB-разъему платы).

void resetProcesses() {
  constGasActive = false;
  manualGasSupplyActive = false;
  piezoActive = false;
  digitalWrite(pinPiezo, LOW);
}

void startPiezo() {
  if (!piezoActive && !isLocked && systemOn) {
    piezoActive = true;
    piezoStartTime = millis();
    digitalWrite(pinPiezo, HIGH);
  }
}

void executeCommand(byte cmd) {
  if (isPurging && cmd != 2) return; 
  switch (cmd) {
    case 1: systemOn = !systemOn; break;
    case 2: isLocked = !isLocked; break;
    case 3: constGasActive = !constGasActive; break;
    case 4: startPiezo(); break;
    case 6: targetPressure = (targetPressure == 1) ? 4 : 1; break;
    case 7: manualGasSupplyActive = !manualGasSupplyActive; break;
    case 99: 
      if (systemOn) {
        isPurging = true;
        purgeStartTime = millis();
        resetProcesses();
      }
      break;
  }
}

void setup() {
  Serial.begin(115200);
  pinMode(pinElectrolyzer, OUTPUT); digitalWrite(pinElectrolyzer, LOW);
  pinMode(pinValve, OUTPUT);        digitalWrite(pinValve, LOW);
  pinMode(pinPiezo, OUTPUT);        digitalWrite(pinPiezo, LOW);
  pinMode(pinButton, INPUT_PULLUP);

  if (radio.begin()) {
    radio.openReadingPipe(0, address);
    radio.setPALevel(RF24_PA_MAX);
    radio.startListening();
  }

  WiFi.softAP(ap_ssid, ap_password);

  server.on("/", HTTP_GET, [](AsyncWebServerRequest *request){
    request->send_P(200, "text/html", index_html);
  });

  server.on("/cmd", HTTP_GET, [](AsyncWebServerRequest *request){
    if (request->hasParam("id")) executeCommand(request->getParam("id")->value().toInt());
    request->send(200, "text/plain", "OK");
  });

  server.on("/status", HTTP_GET, [](AsyncWebServerRequest *request){
    String json = "{\"p\":" + String(currentPressure) + 
                  ",\"pwr\":" + String(systemOn ? 1 : 0) + 
                  ",\"lck\":" + String(isLocked ? 1 : 0) + 
                  ",\"purge\":" + String(isPurging ? 1 : 0) + "}";
    request->send(200, "application/json", json);
  });

  server.begin();
}

void loop() {
  // Чтение давления (10-бит АЦП: 0-1023)
  currentPressure = (analogRead(A0) / 1023.0) * 12.0;

  // Кнопка Power (D0) с двойным кликом
  static bool lastBtnState = HIGH;
  bool btnState = digitalRead(pinButton);
  if (btnState == LOW && lastBtnState == HIGH) {
    unsigned long now = millis();
    if (now - lastClickTime < doubleClickGap) {
      executeCommand(99); // Двойной клик -> Продувка
    } else {
      executeCommand(1);  // Одиночный клик -> Вкл/Выкл
    }
    lastClickTime = now;
    delay(50); 
  }
  lastBtnState = btnState;

  // Радио пульт
  if (radio.available()) {
    radio.read(&dataIn, sizeof(dataIn));
    executeCommand(dataIn.btnId);
  }

  // Логика управления
  unsigned long currentMillis = millis();

  if (isPurging) {
    digitalWrite(pinElectrolyzer, LOW);
    if (currentMillis - purgeStartTime < 5000) {
      digitalWrite(pinValve, HIGH);
    } else {
      digitalWrite(pinValve, LOW);
      isPurging = false;
      systemOn = false;
    }
  } 
  else if (isLocked || !systemOn) {
    digitalWrite(pinElectrolyzer, LOW);
    digitalWrite(pinValve, LOW);
    digitalWrite(pinPiezo, LOW);
    resetProcesses();
  } 
  else {
    bool eNeeded = false;
    bool vNeeded = false;

    if (constGasActive) {
      if (currentPressure < 1.0) { eNeeded = true; vNeeded = true; }
    } else if (manualGasSupplyActive) {
      eNeeded = true; vNeeded = true;
    } else {
      if (currentPressure < targetPressure) eNeeded = true;
    }

    digitalWrite(pinElectrolyzer, eNeeded ? HIGH : LOW);
    digitalWrite(pinValve, vNeeded ? HIGH : LOW);
  }

  // Таймер пьезо
  if (piezoActive && (currentMillis - piezoStartTime >= 500)) {
    piezoActive = false;
    digitalWrite(pinPiezo, LOW);
  }
}

## 🤖 Gemini

Подключиться к твоей NodeMCU очень просто — она сейчас работает как **точка доступа** (как обычный роутер, только без интернета).

Вот пошаговая инструкция:

### 1\. Поиск сети

Возьми смартфон или ноутбук и открой список доступных Wi-Fi сетей. Ты должен увидеть сеть с названием:

-   **Имя (SSID):** `H2_Control_Node`
    
-   **Пароль:** `password123` (если ты не менял его в коде).
    

### 2\. Подключение

Подключись к этой сети.

> **Важно:** Смартфон может выдать предупреждение: *"Эта сеть не имеет доступа к интернету. Оставить подключение?"*. Обязательно нажми **"Да"** или **"Не отключаться"**, иначе телефон автоматически перепрыгнет обратно на домашний Wi-Fi или мобильные данные, и ты не сможешь управлять платой.

### 3\. Вход в панель управления

Когда соединение установлено, открой любой браузер (Chrome, Safari, Opera) и в адресной строке введи IP-адрес платы:

## `192.168.4.1`

После этого должна загрузиться твоя черная панель управления с кнопками и цифрами давления.

---

### Что делать, если не подключается?

1.  **Проверь Serial Monitor:** Подключи плату к компьютеру, открой "Монитор порта" в Arduino IDE (скорость 115200). Если плата перезагружается или пишет ошибки, значит, ей не хватает питания от USB.
    
2.  **Проверь индикацию:** Если на плате NodeMCU горит или мигает синий светодиод, значит, она жива и работает.
    
3.  **Сброс:** Если сеть не появилась, нажми кнопку **RST** на самой плате.
    

### Лайфхак для удобства

Чтобы каждый раз не вводить цифры `192.168.4.1`, можно добавить страницу на "Рабочий стол" телефона (через меню браузера "Добавить на гл. экран"). Тогда она будет открываться как обычное приложение.
